```c
#ifdef DOWNSAMPLE_48k_2_8k
    {
        /* 静态数组避免占栈，大小按 AUDIO_BUFFER_SIZE/4 够用 */
        static double inputBufferD[AUDIO_BUFFER_SIZE / sizeof(int32_t)];
        static double outputD   [AUDIO_BUFFER_SIZE / sizeof(int32_t)];
        float *pF;
        int n_out, j;

        /* 1) float -> double */
        for (j = 0; j < (int)sampleCount; j++) {
            inputBufferD[j] = (double)inputBuffer[j];
        }

        /* 2) 降采样 48k -> 8k */
        n_out = downsample_48k_to_8k(inputBufferD, (int)sampleCount, outputD);

        /* 3) 分配 pFiltered，长度与 sampleCount 一致（DAC 侧仍按 48k 采样） */
        pF = (float *)malloc(sampleCount * sizeof(float));
        if (pF == NULL) return 1u;

        /* 4) 零阶保持：8k -> 48k 扩展回原长度，喂给 DAC */
        for (j = 0; j < (int)sampleCount; j++) {
            int src = j / LPF_Q;
            if (src >= n_out) src = n_out - 1;
            pF[j] = (float)outputD[src];
        }

        pFiltered = pF;
        /* sampleCount 保持不变 */
    }
#endif
```

`lpf_48k_to_8k.h`

```c
#ifndef LPF_48K_TO_8K_H
#define LPF_48K_TO_8K_H

#define LPF_LEN              605
#define LPF_Q                6
#define LPF_GROUP_DELAY      302.0000
#define LPF_GROUP_DELAY_INT  302
.
.
.
static const double lpf_coeff[LPF_LEN] = {...};

int downsample_48k_to_8k(const double *x, int n_in, double *y)
{
    int count = 0;
    int i, k;

    for (i = 0; i < n_in; i += LPF_Q) {     /* 从 0 开始, 不再跳 delay */
        double acc = 0.0;
        for (k = 0; k < LPF_LEN; k++) {
            int idx = i - k;
            if (idx >= 0) acc += lpf_coeff[k] * x[idx];
        }
        y[count++] = acc;
    }
    return count;
}
```
